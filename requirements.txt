# Pinned exactly. The seed generator's output has to be byte-identical between runs,
# which means the interpreter and every library that touches the data are fixed.
# Python 3.12 is the version the step-5 Lambda runs; see .python-version.

psycopg[binary]==3.3.4

# The MCP server. v2 is the line that speaks the stateless protocol:
# the class is MCPServer from mcp.server.mcpserver, and `pip install mcp` resolving to
# 1.x means FastMCP, the initialize handshake and Mcp-Session-Id. Hence the exact pin
# rather than a floor.
mcp==2.0.0

# mcp already depends on uvicorn, but the ASGI server is how this repo runs the server
# locally, so it is a direct dependency here and pinned like one.
uvicorn==0.52.4

# The RDS Data API client, for the aws backend at step 5. The Lambda runtime ships its
# own boto3, so this pin is for the seed script and the bootstrap script running here;
# requirements-lambda.txt deliberately leaves it out of the deployment package.
boto3==1.42.51

# The workbook writer, step 6. It is the only dependency of
# plugin/skills/assemble-delivery-pack/scripts/build_workbook.py, which runs in Cowork's
# code execution sandbox rather than this virtualenv — that script may import openpyxl and
# the standard library and nothing else. The pin here is for scripts/check_pack.py, which
# runs the same script locally and asserts on what it produced.
openpyxl==3.1.5
